#+begin_src emacs-lisp
(require 'json)
(require 'seq)
(require 'cl-lib)
(require 'subr-x)
(let* ((dir (file-name-directory (buffer-file-name)))
       (written nil)
       (avatar-shapes '("blob" "pebble" "bean" "egg" "squircle" "tablet"
                        "capsule" "cylinder" "hex" "gem" "crystal" "wedge"
                        "shield" "dome" "arch" "cloud" "teardrop" "leaf"
                        "square" "sparkle" "clover" "flower" "heart"
                        "house" "star"))
       (avatar-colors '("black" "brown" "red" "orange" "yellow" "green"
                        "cyan" "blue" "violet" "magenta" "gray"))
       (heading-at-point
        (lambda ()
          (substring-no-properties (org-get-heading t t t t))))
       (in-noexport-p
        (lambda ()
          (save-excursion
            (org-back-to-heading t)
            (catch 'found
              (while t
                (when (member "noexport" (org-get-tags nil t))
                  (throw 'found t))
                (unless (org-up-heading-safe)
                  (throw 'found nil)))))))
       (section-instance-field
        (lambda (section)
          (let* ((path (mapcar (lambda (h)
                                 (substring-no-properties (or h "")))
                               (append (org-get-outline-path)
                                       (list (funcall heading-at-point)))))
                 (idx (cl-position section path :test #'cl-equalp)))
            (when (and idx (nth (1+ idx) path) (nth (+ 2 idx) path))
              (list (nth (1+ idx) path) (nth (+ 2 idx) path))))))
       (push-field
        (lambda (cell key text)
          (let ((existing (assoc key (cdr cell))))
            (if existing
                (setcdr existing (concat (cdr existing) "\n\n" text))
              (setcdr cell (append (cdr cell) (list (cons key text))))))))
       (alist-keep
        (lambda (fields keys)
          (let (out)
            (dolist (k keys)
              (let ((v (cdr (assoc k fields))))
                (when (and v (not (string-empty-p (string-trim v))))
                  (push (cons (intern k) (string-trim v)) out))))
            (nreverse out))))
       (require-keys
        (lambda (bot kind label fields keys)
          (dolist (k keys)
            (let ((v (cdr (assoc k fields))))
              (unless (and v (not (string-empty-p (string-trim v))))
                (user-error "%s: %s %S missing %s" bot kind label k)))))))
  (save-excursion
    (save-restriction
      (widen)
      (org-map-entries
       (lambda ()
         (save-restriction
           (org-narrow-to-subtree)
           (let (name description avatar-shape avatar-color memory groups
                      getting-started visibility)
             (org-babel-map-src-blocks nil
               (let ((field (cdr (assq :json (org-babel-parse-header-arguments header-args))))
                     (text (string-trim
                             (if (assq :noweb (org-babel-parse-header-arguments header-args))
                                 ;; Expand <<refs>> against the whole file, not the narrowed bot.
                                 (save-excursion
                                   (save-match-data
                                     (save-restriction
                                       (widen)
                                       (org-babel-expand-noweb-references
                                        (org-babel-get-src-block-info)))))
                               (org-unescape-code-in-string (or body ""))))))
                 (when (and field
                            (not (string-empty-p text))
                            (not (funcall in-noexport-p)))
                   (pcase (format "%s" field)
                     ("name" (setq name text))
                     ("description" (push text description))
                     ("avatarShape" (setq avatar-shape text))
                     ("avatarColor" (setq avatar-color text))
                     ("gettingStarted" (setq getting-started text))
                     ("visibility" (setq visibility text))
                     ("memory"
                      (let* ((h (funcall heading-at-point))
                             (kind (cond
                                    ((cl-equalp h "profile") "profile")
                                    ((cl-equalp h "log") "log")
                                    (t nil))))
                        (dolist (line (split-string text "\n" t "[ \t]+"))
                          (push (if kind
                                    `((content . ,line) (kind . ,kind))
                                  `((content . ,line)))
                                memory))))

                      (let ((place (funcall section-instance-field section)))
                        (unless place
                          (user-error "%s: a :json %s block is not under a field heading"
                                      (or name (funcall heading-at-point))
                                      section))

                               (key (cadr place))
                               (id (cons section instance))
                               (cell (assoc id groups)))

                            (push cell groups))

                 (setq avatar-color (string-trim avatar-color))
                 (unless (member avatar-color avatar-colors)
                   (user-error "%s: avatarColor %S is not allowed (%s)"
                               name avatar-color (string-join avatar-colors " "))))
               (setq description (string-join (nreverse description) "\n\n"))
               (when (string-empty-p (or description ""))
                 (user-error "%s: profile is missing a description" name))
               (when visibility
                 (setq visibility (string-trim visibility))
                 (unless (member visibility '("public" "team"))
                   (user-error "%s: visibility %S is not allowed (public, team)"
                               name visibility)))
               (when getting-started
                 (setq getting-started (string-trim getting-started)))
               (let* ((skill-names nil)
                      (objects
                       (lambda (section required keys)
                         (let (acc)
                           (dolist (g (reverse groups))
                             (when (equal (caar g) section)
                               (let* ((instance (cdar g))
                                      (fields (cdr g)))
                                 (when (equal section "routines")
                                   (let* ((schedule (string-trim (or (cdr (assoc "schedule" fields)) "")))
                                          (content (cdr (assoc "content" fields))))
                                     (setq fields (cl-remove "schedule" fields :test #'equal :key #'car))
                                     (unless (string-empty-p schedule)
                                       (let ((line (format "Schedule: %s (local time)" schedule)))
                                         (when content
                                           (setcdr (assoc "content" fields)
                                                   (concat line "\n" content)))))))
                                 (funcall require-keys
                                          name
                                          (pcase section
                                            ("skills" "skill")
                                            ("routines" "routine")
                                            ("plugins" "plugin"))
                                          (or (cdr (assoc (pcase section
                                                            ("skills" "name")
                                                            ("routines" "slug")
                                                            ("plugins" "pluginId"))
                                                          fields))
                                              instance)
                                          fields
                                          required)
                                 (when (equal section "skills")
                                   (push (string-trim (cdr (assoc "name" fields)))
                                         skill-names))
                                 (push (funcall alist-keep fields keys) acc))))
                           (vconcat (nreverse acc)))))
                      (skills (funcall objects "skills"
                                       '("name" "content")
                                       '("name" "content" "description")))
                      (routines (funcall objects "routines"
                                         '("slug" "description" "content")
                                         '("slug" "description" "content" "name")))
                      (plugins (funcall objects "plugins"
                                        '("pluginId")
                                        '("pluginId" "name" "description")))
                      (file (concat (replace-regexp-in-string
                                     "[^a-z0-9]+" "-" (downcase name))
                                    ".json"))
                      (profile `((name . ,name)
                                 (description . ,description)))
                      (json-encoding-pretty-print t)
                      (coding-system-for-write 'utf-8)
                      obj)
                 (when getting-started
                   (unless (member getting-started skill-names)
                     (user-error "%s: gettingStarted names missing skill %S"
                                 name getting-started)))
                 (when avatar-shape
                   (setq profile (append profile (list (cons 'avatarShape avatar-shape)))))
                 (when avatar-color
                   (setq profile (append profile (list (cons 'avatarColor avatar-color)))))
                 (setq obj `((profile . ,profile)
                             (memory . ,(vconcat (nreverse memory)))
                             (skills . ,skills)
                             (routines . ,routines)
                             (plugins . ,plugins)))
                 (when getting-started
                   (setq obj (append obj (list (cons 'gettingStarted
                                                     (list (cons 'skill getting-started)))))))
                 (when visibility
                   (setq obj (append obj (list (cons 'visibility visibility)))))
                 (when (member file written)
                   (user-error "Two bots tangle to %s" file))
                 (push file written)
                 (with-temp-file (expand-file-name file dir)
                   (insert (json-encode obj) "\n")))))))
       "LEVEL=1-noexport")))
  (message "Wrote %s" (if written (string-join (reverse written) ", ") "no bot files"))
  written)
