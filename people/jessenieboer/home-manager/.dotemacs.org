** controlling emacs by keyboard
:PROPERTIES:
:NAME:     keyboard-controls
:DEPENDS:  environment-non-nix-packages
:ID:       605f98ee-769b-4678-995b-66234bd364c7
:END:
F1 for control mode, F7 for literal mode. Controls will change based on buffer's major mode.

Keep mode lists separate rather than put them directly in the my-hydra-define call to keep that call cleaner and so that I can fold away these big lists of modes.

4 general kinds of controls:
- connection: controlling what files and processes (buffers) *emacs* is connected to, and which ones are currently in front of you to work with (the ones *you* are currently "connected to")
- navigation: controlling where your cursor is on the screen
- display: controlling where things are on screen and how they look
- interaction: controlling what you are doing with the buffer you're currently working with

when figuring out what you want keys to be, consider
- mnemonic
- physical (what I use)

the control key is for enabling an extra layer on top of modes that need normal text editing capabilities like org and shell
#+begin_src emacs-lisp
(use-package major-mode-hydra
  :config
  (setq hydra-is-helpful nil))

;; make it cleaner to define hydras by being able to separate the list of modes from the definition of the keys rather than having them all in the same call
(defmacro my-make-hydra (modes keys)
  `(major-mode-hydra-define ,(eval modes) (:color amaranth :quit-key ("<f7>") :title "Controls") ,keys))

(defmacro my-add-to-hydra (modes keys)
  `(major-mode-hydra-define+ ,(eval modes) (:color amaranth :quit-key ("<f7>") :title "Controls") ,keys))
(setq my-recorded-major-mode nil)

(defun my-record-major-mode ()
  (setq my-recorded-major-mode major-mode))

(defun my-autohydra ()
  "Switches to the appropriate hydra for the buffer when focus changes"
  (if (not (or (eq major-mode 'minibuffer-mode)
  	       (eq major-mode 'minibuffer-inactive-mode)
	       (eq major-mode 'fundamental-mode) ;; for org capture templates
	       (eq my-recorded-major-mode 'fundamental-mode) ;; for org capture templates
  	       (eq my-recorded-major-mode 'minibuffer-mode)
  	       (eq my-recorded-major-mode 'minibuffer-inactive-mode)))
      (setq unread-command-events '(f7 f1))))

(defun my-show-major-mode ()
  "Displays the name of the major mode at the bottom of the screen. Makes it easier to see what modes I need to add to various hydras."
  (interactive)
  (message "major-mode: %s" major-mode))

(use-package buffer-focus-hook

  (defun my-dir-within-p (dir parent)
    "Return t if DIR is within PARENT, nil otherwise."
    (let ((dir (expand-file-name dir))
          (parent (expand-file-name parent)))
      (string-prefix-p (file-name-as-directory parent)
                       (file-name-as-directory dir))))

  (defvar my-locate-dominating-file-cache (make-hash-table :test 'equal))

  (defun my-locate-dominating-file (dir marker)
    "Optimized version of locate-dominating-file with caching."
    (or (gethash dir my-locate-dominating-file-cache)
        (let ((result (locate-dominating-file dir marker)))
          (puthash dir result my-locate-dominating-file-cache)
          result)))

  (defun my-find-project-containing-dir (dir)
    "Find the nearest ancestor directory containing a project marker file.
        Return the (transient . \"directory path\"), or nil if nothing is found.
        Used in `project-find-functions`. Ignores directories not under
        `my-projects-directory` and any path containing `/.direnv/`.
        Accepts both file and directory paths as input."

- repository :: a collection of materials that share an overarching categorization that is mutually exclusive with other overarching categorizations, and thus are stored under one top-level container and can have a heirarchy structure. implemented as a directory.
- project :: an ordering of resources (labor, money, materials, etc...) aiming at some defined or definable benefit. project materials are contained in repositories.
- project type :: some higher-level categorization of projects.
- workplace :: any organization, like a company or my household. contains workshops. implemented as a github repo with a nix flake that defines each workshop as a nixos configuration
- workshop :: the entire computer environment including hardware, os, software, and generic configurations. described with nix. one active workshop per physical machine, but machines can switch between workshops. contains worksbenches. implemented in the workplace repo as a single nixos config
- toolbox :: a set of tools and configurations aimed at a specific kind of work. one wb for general development, one for python development, etc... implemented as a template repo with a nix flake containing  dev shell(s)? workplaces, workshops, and/or projects may provide toolboxes.
- toolbelt :: someone's personal set of tools and configurations. implemented with a flake and home manager. might have a separate repo for nix-independent emacs?  
- workspace :: personal "area" within the workshop, basically your account on the desktop?
- workbench :: a personally-constructed tool-specific arrangement of tools and materials, where work is ready to go. shorter-term focus; may flit around between workbenches. os workbenches are probably "desktops". emacs workbenches are flexible and involve
  - emacs frame arrangements
  - emacs window arrangements
  - the buffer groups available to windows
    - use tab-bar, tab-line, and emacs purpose?
      https://github.com/bmag/emacs-purpose
      perhaps activities is in there too https://github.com/alphapapa/activities.el
- screen :: a physical display
  
emacs terms:
- frame :: container for windows. 
- tab bar :: used to contain workbenches
- buffer :: connection to a computer file or process
- buffer group :: an collection of buffers organized by name, mode, project, etc...
- minibuffer :: buffer that connects to emacs itself?
- tab :: convnient way of displaying and selecting buffers in a window
- mode :: rules for how a buffer behaves
- window :: displays a buffer and can be limited to show only a specific group of buffers
- side window ::
- workspace/zone/station, project zone :: a conceptual area within a workshop arranged to facilitate some scope of work within or among projects. workspaces are flexible and defined on the fly, but saveable. they include screen setup; perhaps managed at the os level?

3 frame:
- middle frame "main": primary interaction
- left frame "references" (info useful for main frame) and "input" (command line): 
- right frame "results" or "output": output of main


tab grouping:
- if a buffer is in a project, that's its tab group, "No project" otherwise
- some tabs are hidden from tab line altogether

  (defun my-determine-category (effort-type frequency)
    (cond ((and (string= effort-type "work")
  	      (string= frequency "once")) "task")
  	((and (string= effort-type "work")
  	      (string= frequency "repetitive")) "routine")
  	((and (string= effort-type "work")
  	      (string= frequency "continuous")) "practice")
  	((and (string= effort-type "none")
  	      (string= frequency "once")) "occurance")
  	((and (string= effort-type "none")
  	      (string= frequency "repetitive")) "recurrence")
  	((and (string= effort-type "organization")
  	      (string= frequency "once")) "reference")))

  	(org-entry-put nil "CATEGORY" (my-determine-category effort-type frequency)))))

(defun my-org-cmp-relative-value (a b property)
  "Compare agenda entries by less/average/more. 
Returns +1 if A > B, -1 if A < B, nil if equal or missing."
  (if (and a b)
      (let* ((ea (my-get-agenda-entry-property a property))
             (eb (my-get-agenda-entry-property b property))
             (sea (or ea "less"))
             (seb (or eb "less"))
             (order '("less" "average" "more"))
             (ia (cl-position sea order :test #'string=))
             (ib (cl-position seb order :test #'string=)))
        (cond ((= ia ib) nil)
              ((< ia ib) -1)
              ((> ia ib) +1)))
    nil))

(defun my-org-src-setup-for-lsp (modes)
  "Set up temporary file for org-src buffers in MODES with LSP support.
  MODES is a list of major modes (e.g., '(prolog-mode sweeprolog-mode))."
  (when (apply #'derived-mode-p modes)
    ;; make a temp file in same dir
    (let ((org-file (buffer-file-name (org-src-source-buffer))))
      (when org-file
        (let ((temp-file (expand-file-name
                          (concat (file-name-base org-file)
                                  "-derived-"
                                  (format "%06x" (random (expt 16 6)))
                                  ".pl")
                          (file-name-directory org-file))))
          (setq-local buffer-file-name temp-file)
          (write-region (point-min) (point-max) temp-file nil 'silent))))
    ;; Mark buffer as unmodified to keep org-src-exit ability
    ;;(set-buffer-modified-p nil)
    ;; start lsp
    (lsp)))

(defun my-org-edit-src-exit-advice (orig-fun &rest args)
  "Clean up temporary file before exiting org-edit-src buffer."
  (let ((temp-file buffer-file-name))
    (unwind-protect
        (apply orig-fun args)
      (when (and temp-file (file-exists-p temp-file))
        (delete-file temp-file)))))

(advice-add 'org-edit-src-exit :around #'my-org-edit-src-exit-advice)

(defun my-org-agenda-cycle-span (direction)
  "Cycle agenda time span forward or backward.
DIRECTION: +1 for forward (day → week → month → year), -1 for backward.
Stops at ends, no wraparound."
  (interactive "p")
  (unless (derived-mode-p 'org-agenda-mode)
    (user-error "Not in an org-agenda buffer"))
  (let* ((current org-agenda-current-span)
         (spans '(day week month year))
         (pos (cl-position current spans))
         (new-pos (max 0 (min (+ pos direction) (1- (length spans)))))
         (new-span (nth new-pos spans)))
    (if (= pos new-pos)
        (message "Already at %s span: %s" (if (> direction 0) "maximum" "minimum") current)
      (org-agenda-change-time-span new-span)
      (message "Span: %s → %s" current new-span))))

  (interactive)
  (when (eq major-mode 'org-mode)
    (let* ((org-file (buffer-file-name))
           (md-file
            (or

             (org-entry-get nil "EXPORT_FILE_NAME" t)

          (grip-start-preview md-file-full)

  "Send highlighted region or goal at point to sweeprolog top-level."
  (interactive "r")

                   (buffer-substring-no-properties start end)
                 (or (car (sweeprolog-goals-at-point))
                     (error "No goal at point"))))
         (goal (cond

         (buffer (sweeprolog-top-level-buffer)))

    (display-buffer buffer sweeprolog-top-level-display-action)))
