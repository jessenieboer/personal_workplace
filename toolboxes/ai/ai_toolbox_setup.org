  (defun ai-setup-mcp-servers ()
  (when-let ((cmd (executable-find "mcp-server-filesystem")))
    (setq-local mcp-hub-servers   ; ← prefer buffer-local if possible
                `(("filesystem" . (:command ,cmd
                                    :args (".")
                                    :roots (,(project-root (project-current)))))))))

  (defun ai-setup-agents ()
    (setq gptel-agent-dirs (list (expand-file-name ".toolboxes/agents/" (project-root (project-current)))))
    (gptel-agent-update)
    ;; add the vizier agent as a preset so that gptel can use it as the default agent
    (when-let* ((gptel-agent-plist (assoc-default "ai_vizier" gptel-agent--agents nil nil)))
    (apply #'gptel-make-preset 'ai_vizier gptel-agent-plist)))

   (let ((root (project-root (project-current))))
  (message "setting up ai for %s" (project-name (project-current)))
  (ai-setup-agents)
  (ai-setup-mcp-servers)
  (message "ai for %s ready" (project-name (project-current))))
#+END_SRC
